struct node {
    int coeff, exp;
    struct node *next;
};

struct node *create(int c, int e) {
    struct node *t;
    t = (struct node *)malloc(sizeof(struct node));
    t->coeff = c;
    t->exp = e;
    t->next = NULL;
    return t;
}

struct node *insert(struct node *head, int c, int e) {
    struct node *t, *p;
    t = create(c, e);

    if (head == NULL)
        return t;

    p = head;
    while (p->next != NULL)
        p = p->next;

    p->next = t;
    return head;
}

struct node *readPoly() {
    struct node *head = NULL;
    int n, i, c, e;

    printf("Enter number of terms: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++) {
        printf("Enter coefficient and exponent: ");
        scanf("%d%d", &c, &e);
        head = insert(head, c, e);
    }

    return head;
}

void display(struct node *p) {
    while (p != NULL) {
        printf("%dx^%d", p->coeff, p->exp);
        if (p->next != NULL)
            printf(" + ");
        p = p->next;
    }
    printf("\n");
}

struct node *calculate(struct node *a,
                       struct node *b, int sign) {
    struct node *result = NULL;

    while (a != NULL && b != NULL) {
        if (a->exp == b->exp) {
            if (a->coeff + sign * b->coeff != 0)
                result = insert(result,
                    a->coeff + sign * b->coeff, a->exp);
            a = a->next;
            b = b->next;
        } else if (a->exp > b->exp) {
            result = insert(result, a->coeff, a->exp);
            a = a->next;
        } else {
            result = insert(result, sign * b->coeff, b->exp);
            b = b->next;
        }
    }

    while (a != NULL) {
        result = insert(result, a->coeff, a->exp);
        a = a->next;
    }

    while (b != NULL) {
        result = insert(result, sign * b->coeff, b->exp);
        b = b->next;
    }

    return result;
}

    struct node dummy;
    struct node *tail = &dummy;

    dummy.bottom = NULL;

    while (a != NULL && b != NULL) {
        if (a->data <= b->data) {

            a = a->bottom;

            b = b->bottom;

        tail = tail->bottom;
        tail->next = NULL;

        tail->bottom = a;
    else
        tail->bottom = b;

    return dummy.bottom;
}

    struct node *t = create(x);

    if (head == NULL)
