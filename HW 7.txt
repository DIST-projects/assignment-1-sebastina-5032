struct Node* createNode(char data) {
    struct Node *newNode = malloc(sizeof(struct Node));
    newNode->data = data;
    newNode->left = newNode->right = NULL;
    return newNode;
}

void inorder(struct Node *root) {
    if (root != NULL) {
        if (root->left || root->right)
            printf("(");
        inorder(root->left);
        printf("%c", root->data);
        inorder(root->right);
        if (root->left || root->right)
            printf(")");
    }
}

    root->left->left = createNode('a');
    root->left->right = createNode('b');
    root->right->left = createNode('c');
    root->right->right = createNode('d');

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node* createTree() {
    int x;
    struct Node *newNode;

    scanf("%d", &x);
    if (x == -1)
        return NULL;

    newNode = malloc(sizeof(struct Node));
    newNode->data = x;
    newNode->left = createTree();
    newNode->right = createTree();

    return newNode;
}

int isBST(struct Node *root, int min, int max) {
    if (root == NULL)
        return 1;
    if (root->data <= min || root->data >= max)
        return 0;

    return isBST(root->left, min, root->data) &&
           isBST(root->right, root->data, max);
}

int main() {
    struct Node *root;

    printf("Enter tree nodes (-1 for NULL): ");
    root = createTree();

    if (isBST(root, INT_MIN, INT_MAX))
        printf("Binary Search Tree");
    else
        printf("Not a Binary Search Tree");

    return 0;
}

int a[50], n;

void heapify(int i) {
    int largest = i, l = 2*i+1, r = 2*i+2, t;

    if (l < n && a[l] > a[largest])
        largest = l;
    if (r < n && a[r] > a[largest])
        largest = r;

    if (largest != i) {
        t = a[i];
        a[i] = a[largest];
        a[largest] = t;
        heapify(largest);
    }
}

int main() {
    int i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    for (i = n/2-1; i >= 0; i--)
        heapify(i);

    printf("Max Heap: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

}

int parent[20];

int find(int i) {
    while (parent[i] != i)
        i = parent[i];
    return i;
}

int main() {
    int a[20][20], n, i, j, u, v, min, cost = 0, edges = 0;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    printf("Enter adjacency matrix (0 for no edge):\n");
    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            scanf("%d", &a[i][j]);

    for (i = 0; i < n; i++)
        parent[i] = i;

    printf("Edges in MST:\n");

    while (edges < n - 1) {
        min = 9999;
        u = v = -1;

        for (i = 0; i < n; i++)
            for (j = i + 1; j < n; j++)
                if (a[i][j] > 0 && a[i][j] < min) {
                    min = a[i][j];
                    u = i;
                    v = j;
                }

        if (u == -1)
            break;

        i = find(u);
        j = find(v);

        if (i != j) {
            printf("%d - %d : %d\n", u, v, min);
            cost += min;
            parent[i] = j;
            edges++;
        }

        a[u][v] = a[v][u] = 0;
    }

    if (edges == n - 1)
        printf("Minimum cost = %d\n", cost);
    else
        printf("MST not possible\n");

    return 0;
}

int a[20][20], color[20], n;

int bipartite(int v, int c) {
    int i;
    color[v] = c;

    for (i = 0; i < n; i++) {
        if (a[v][i]) {
            if (color[i] == -1) {
                if (!bipartite(i, 1 - c))
                    return 0;
            }
            else if (color[i] == c)
                return 0;
        }
    }
    return 1;
}

int main() {
    int i, j;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

        for (j = 0; j < n; j++)

    for (i = 0; i < n; i++)
        color[i] = -1;

    for (i = 0; i < n; i++)
        if (color[i] == -1 && !bipartite(i, 0)) {
            printf("Not Bipartite");
            return 0;
        }

    printf("Bipartite Graph");
    return 0;
}
