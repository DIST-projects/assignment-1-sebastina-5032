void swap(int x, int y) {
    struct node *a = head, *b = head, *t;

    while (a && a->data != x) a = a->next;
    while (b && b->data != y) b = b->next;

    if (!a || !b || a == b) return;

    if (a->next == b) { t = a; a = b; b = t; }

    if (b->next == a) {
        t = a; a = b; b = t;
    }

    if (a->next == b) {
        a->next = b->next;
        b->prev = a->prev;
        b->next = a;
        a->prev = b;
        if (b->prev) b->prev->next = b;
        else head = b;
        if (a->next) a->next->prev = a;
        return;
    }

    if (a->prev) a->prev->next = b;
    else head = b;
    if (b->prev) b->prev->next = a;
    else head = a;

    if (a->next) a->next->prev = b;
    if (b->next) b->next->prev = a;

    t = a->prev; a->prev = b->prev; b->prev = t;
    t = a->next; a->next = b->next; b->next = t;
}

void display() {
    struct node *t = head;
    while (t) {
        printf("%d ", t->data);
        t = t->next;
    }
    printf("\n");
}

    struct node *q[100], *t;
    int front = 0, rear = 0, count, i;

    if (!root) return;
    q[rear++] = root;

    while (front < rear) {
        count = rear - front;

        for (i = 0; i < count; i++) {
            t = q[front++];
            if (i < count - 1)
                t->nextRight = q[front];

    }

void heapify(int a[], int n, int i)
{
    int largest = i, l = 2 * i + 1, r = 2 * i + 2, temp;

    if (l < n && a[l] > a[largest])
        largest = l;

    if (r < n && a[r] > a[largest])
        largest = r;

    if (largest != i)
    {
        temp = a[i];
        a[i] = a[largest];
        a[largest] = temp;
        heapify(a, n, largest);
    }
}

void heapSort(int a[], int n)
{
    int i, temp;

    for (i = n / 2 - 1; i >= 0; i--)
        heapify(a, n, i);

    for (i = n - 1; i > 0; i--)
    {
        temp = a[0];
        a[0] = a[i];
        a[i] = temp;
        heapify(a, i, 0);
    }
}
