    int inches;
};

{

    for(i = 0; i < n; i++)
    {
        scanf("%d %d", &d[i].feet, &d[i].inches);
    }

    for(i = 0; i < n; i++)
    {
        sum.feet += d[i].feet;
        sum.inches += d[i].inches;
    }

    sum.feet += sum.inches / 12;
    sum.inches = sum.inches % 12;

    return 0;

struct Time
{
    int hours;
    int minutes;
    int seconds;
};

int main()
{
    struct Time t1, t2, diff;

    printf("Enter first time (hours minutes seconds): ");
    scanf("%d %d %d", &t1.hours, &t1.minutes, &t1.seconds);

    printf("Enter second time (hours minutes seconds): ");
    scanf("%d %d %d", &t2.hours, &t2.minutes, &t2.seconds);


    if(t1.seconds < t2.seconds)
    {
        t1.seconds += 60;
        t1.minutes--;
    }

 
    if(t1.minutes < t2.minutes)
    {
        t1.minutes += 60;
        t1.hours--;
    }

    diff.seconds = t1.seconds - t2.seconds;
    diff.minutes = t1.minutes - t2.minutes;
    diff.hours = t1.hours - t2.hours;

    printf("Difference = %d hours %d minutes %d seconds",
           diff.hours, diff.minutes, diff.seconds);

    return 0;
}

float areoftri(int b, int h){
    return 1.0/2*b*h;
}

float areaofrect(int l, int b){
    return l*b;
}

int peroftri(int a, int b, int c){
    return a+b+c;
}

int perofrect(int l, int b){
    return 2*(l+b);
}

int main(){
    int l, b, ba, h, a, c;
    
    float (*f1)(int, int);
    float (*f2)(int, int);
    int (*f3)(int, int, int);
    int (*f4)(int, int);

    f1 = areoftri;
    f2 = areaofrect;
    f3 = peroftri;
    f4 = perofrect;

    printf("Enter length and breadth of rectangle: ");
    scanf("%d %d", &l, &b);

    printf("Area of rectangle = %.2f\n", f2(l, b));
    printf("Perimeter of rectangle = %d\n", f4(l, b));

    printf("Enter base and height of triangle: ");
    scanf("%d %d", &ba, &h);

    printf("Area of triangle = %.2f\n", f1(ba, h));

    printf("Enter three sides of triangle: ");
    scanf("%d %d %d", &a, &b, &c);

    printf("Perimeter of triangle = %d\n", f3(a, b, c));

    return 0;
}
