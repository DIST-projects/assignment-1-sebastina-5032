1)#include <stdio.h>
#define MAX 20

int q1[MAX], q2[MAX];
int f1 = 0, r1 = 0, f2 = 0, r2 = 0;

void push(int x)
{
    while (f1 < r1)
        q2[r2++] = q1[f1++];

    q1[r1++] = x;

    while (f2 < r2)
        q1[r1++] = q2[f2++];

    f1 = 0;
    r2 = 0;
    f2 = 0;
}

int pop()
{
    if (f1 == r1)
        return -1;

    return q1[f1++];
}

void display()
{
    int i;

    for (i = f1; i < r1; i++)
        printf("%d ", q1[i]);
}

int main()
{
    int n, x, i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter stack elements:\n");
    for (i = 0; i < n; i++)
    {
        scanf("%d", &x);
        push(x);
    }

    printf("Stack elements: ");
    display();

    printf("\nPopped element: %d", pop());

    printf("\nStack after pop: ");
    display();

    return 0;
}

#include <stdio.h>
#define MAX 20

int s1[MAX], s2[MAX];
int top1 = -1, top2 = -1;

void enqueue(int x)
{
    s1[++top1] = x;
}

int dequeue()
{
    int x;

    if (top1 == -1 && top2 == -1)
        return -1;

    if (top2 == -1)
    {
        while (top1 != -1)
            s2[++top2] = s1[top1--];
    }

    x = s2[top2--];
    return x;
}

void display()
{
    int i;

    if (top2 != -1)
        for (i = top2; i >= 0; i--)
            printf("%d ", s2[i]);

    for (i = 0; i <= top1; i++)
        printf("%d ", s1[i]);
}

int main()
{
    int n, x, i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter queue elements:\n");
    for (i = 0; i < n; i++)
    {
        scanf("%d", &x);
        enqueue(x);
    }

    printf("Queue elements: ");
    display();

    printf("\nDeleted element: %d", dequeue());

    printf("\nQueue after deletion: ");
    display();

    return 0;
}

#include <stdio.h>
#include <string.h>
#define MAX 20

char back[MAX][50], forward[MAX][50];
int topBack = -1, topForward = -1;
char current[50] = "Home";

void visit(char page[])
{
    strcpy(back[++topBack], current);
    strcpy(current, page);
    topForward = -1;
}

void goBack()
{
    if (topBack == -1)
    {
        printf("\nNo previous page.");
        return;
    }

    strcpy(forward[++topForward], current);
    strcpy(current, back[topBack--]);
}

void goForward()
{
    if (topForward == -1)
    {
        printf("\nNo forward page.");
        return;
    }

    strcpy(back[++topBack], current);
    strcpy(current, forward[topForward--]);
}

int main()
{
    visit("Google");
    visit("YouTube");
    visit("Wikipedia");

    printf("Current page: %s", current);

    goBack();
    printf("\nAfter Back: %s", current);

    goBack();
    printf("\nAfter Back: %s", current);

    goForward();
    printf("\nAfter Forward: %s", current);

    return 0;
}
Output:

After Back: YouTube
After Back: Google
After Forward: YouTube

            if (arr[i] > arr[j])

                temp = arr[i];

#include <stdio.h>
#define MAX 20

struct Cell
{
    int r, c, time;
};

int main()
{
    int a[MAX][MAX], m, n, i, j;
    int front = 0, rear = -1;
    int fresh = 0, time = 0;
    struct Cell q[MAX * MAX];

    printf("Enter rows and columns: ");
    scanf("%d %d", &m, &n);

    printf("Enter matrix:\n");

    for (i = 0; i < m; i++)
    {
        for (j = 0; j < n; j++)
        {
            scanf("%d", &a[i][j]);

            if (a[i][j] == 1)
                fresh++;

            if (a[i][j] == 2)
            {
                q[++rear].r = i;
                q[rear].c = j;
                q[rear].time = 0;
            }
        }
    }

    while (front <= rear)
    {
        struct Cell cur = q[front++];
        int dr[4] = {-1, 1, 0, 0};
        int dc[4] = {0, 0, -1, 1};

        if (cur.time > time)
            time = cur.time;

        for (i = 0; i < 4; i++)
        {
            int nr = cur.r + dr[i];
            int nc = cur.c + dc[i];

            if (nr >= 0 && nr < m && nc >= 0 && nc < n &&
                a[nr][nc] == 1)
            {
                a[nr][nc] = 2;
                fresh--;

                q[++rear].r = nr;
                q[rear].c = nc;
                q[rear].time = cur.time + 1;
            }
        }
    }

    if (fresh == 0)
        printf("Minimum time = %d", time);
    else
        printf("All oranges cannot be rotten.");

    return 0;
}
Output:
Enter rows and columns: 3 3
Enter matrix:

1 0 2
0 0 2
Minimum time = 1
