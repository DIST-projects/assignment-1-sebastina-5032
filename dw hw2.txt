#include<stdio.h>

int smallest(float a[], int k, int n)
{
    int i, pos = k;
    float small = a[k];

    for(i = k + 1; i < n; i++)
    {
        if(a[i] < small)
        {
            small = a[i];
            pos = i;
        }
    }

    return pos;
}

void sel_sort(float a[], int n)
{
    int  k, pos;
    float temp;

    for(k = 0; k < n - 1; k++)
    {
        pos = smallest(a, k, n);

        temp = a[k];
        a[k] = a[pos];
        a[pos] = temp;
    }
}

int main()
{
    int i, n;
    scanf("%d",&n);

    float a[n];

    for(i = 0; i < n; i++)
        scanf("%f",&a[i]);

    sel_sort(a,n);

    printf("Sorted array:\n");

    for(i = 0; i < n; i++)
        printf("%.2f ",a[i]);

    return 0;
}

struct Student
{
    int id;
    char name[20];
    char grade;
};

void sort(struct Student a[], int n, int choice)
{
    int i,j;
    struct Student temp;

    for(i=0;i<n-1;i++)
    {
        for(j=i+1;j<n;j++)
        {
            if((choice==1 && a[i].id>a[j].id) ||
               (choice==2 && strcmp(a[i].name,a[j].name)>0) ||
               (choice==3 && a[i].grade>a[j].grade))
            {
                temp=a[i];
                a[i]=a[j];
                a[j]=temp;
            }
        }
    }
}

int main()
{
    int i,n,choice;

    scanf("%d",&n);

    struct Student a[n];

    for(i=0;i<n;i++)
        scanf("%d %s %c",&a[i].id,a[i].name,&a[i].grade);

    printf("Select sorting criteria:\n");
    printf("1-ID  2-Name  3-Grade\n");
    scanf("%d",&choice);

    sort(a,n,choice);

    for(i=0;i<n;i++)
        printf("%d %s %c\n",a[i].id,a[i].name,a[i].grade);

    return 0;
}

9)#include<stdio.h>

void radix(int a[], int n)
{
    int bucket[10][20];
    int bucket_count[10];
    int i,j,k,digit,div=1,max=a[0];

    for(i=1;i<n;i++)
        if(a[i]>max)
            max=a[i];

    while(max/div>0)
    {
        for(i=0;i<10;i++)
            bucket_count[i]=0;

        for(i=0;i<n;i++)
        {
            digit=(a[i]/div)%10;
            bucket[digit][bucket_count[digit]]=a[i];
            bucket_count[digit]++;
        }

        k=0;

        for(i=0;i<10;i++)
        {
            for(j=0;j<bucket_count[i];j++)
            {
                a[k]=bucket[i][j];
                k++;
            }
        }

        div=div*10;
    }
}

int main()
{
    int a[20],pos[20],neg[20];
    int n,i,p=0,k=0,j=0;

    scanf("%d",&n);

    for(i=0;i<n;i++)
        scanf("%d",&a[i]);



    for(i=0;i<n;i++)
    {
        if(a[i]<0)
            neg[k++]=-a[i];
        else
            pos[p++]=a[i];
    }


    radix(neg,k);
    radix(pos,p);


    for(i=k-1;i>=0;i--)
        a[j++]=-neg[i];


    for(i=0;i<p;i++)
        a[j++]=pos[i];

    printf("Sorted array:\n");

    for(i=0;i<n;i++)
        printf("%d ",a[i]);

    return 0;
}
